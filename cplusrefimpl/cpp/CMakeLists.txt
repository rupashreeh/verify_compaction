cmake_minimum_required(VERSION 3.14)
project(lsm_wal_memtable_sst CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(GLOB LSM_SRC src/*.cc)
add_library(lsm STATIC ${LSM_SRC})
target_include_directories(lsm PUBLIC src)
enable_testing()
foreach(t basic crash bugs random)
  add_executable(test_${t} tests/test_${t}.cc)
  target_include_directories(test_${t} PRIVATE tests)
  target_link_libraries(test_${t} lsm)
  add_test(NAME ${t} COMMAND test_${t})
endforeach()
